\section{Objective 1: Dominant Mode Identification}

% --- Divider: Objective 1 ---
\begin{frame}
    \vspace{0.3cm}
    \begin{center}
        {\usebeamerfont{title}\structure{\Huge Objective 1}}\\[0.3cm]
        {\Large Dominant Mode Identification}
    \end{center}
    \vspace{0.7cm}
    \begin{itemize}
        \item Identify the dominant coherent MHD mode(s) present in the Mirnov coil array using a dual criterion: amplitude above a $3\sigma$ noise threshold \emph{and} spatial coherence $\gamma^2 > 0.7$ across at least one coil pair.
        \item Determine the toroidal mode number $n$ and localize the fluctuation radially using the ECE and PMP diagnostics.
        \item Repeat the analysis separately for the ECH+NBI and NBI-only phases of shot \#88653, to check whether mode identity and localization survive the loss of ECH heating.
    \end{itemize}
\end{frame}

% Slide: Obj 1 Parameters
\begin{frame}{Objective 1: Signal Processing Parameters}
    \textbf{Time-Frequency Analysis Settings (Shot \#88653)}
    \begin{itemize}
        \item \textbf{FFT Spectrogram:} Hann window applied, segment length $T = \SI{2.00}{\milli\second}$, achieving a frequency resolution of $\Delta f = \SI{500.0}{\hertz}$.
        \item \textbf{Wavelet Transform:} Morlet continuous wavelet transform (CWT) evaluated across 150 frequencies (\SI{10}{\kilo\hertz} -- \SI{150}{\kilo\hertz}).
        \item \textbf{Noise Baseline:} Calibrated between \SI{150}{\milli\second} and \SI{175}{\milli\second}.
        \item \textbf{Validation:} Dominant modes must exceed a $3\sigma$ threshold above the baseline and demonstrate spatial coherence ($\gamma^2 > 0.7$) across at least one coil pair.
    \end{itemize}
\end{frame}

% Slide: Obj 1 Dominant Mode Identification
\begin{frame}{Objective 1: Dominant Mode Identification}
    \textbf{Confirmed Dominant and Coherent Modes (Shot \#88653)}
    \begin{itemize}
        \item \textbf{\SI{89.0}{\kilo\hertz} Mode:} Passed dual-criterion test with robust spatial coherence across all pairs (MP1-MP3: 0.855, MP1-MP4: 0.876, MP3-MP4: 0.878).
        \item \textbf{\SI{95.5}{\kilo\hertz} Mode:} Passed dual-criterion test, confirmed by MP3-MP4 coherence (0.710).
        \item \textbf{Observation:} Both modes initiate sharply just after \SI{175}{\milli\second}, consistent with a rapid MHD onset during the discharge.
    \end{itemize}
\end{frame}

% Slide: Re-evaluating Objective 1 Modes
\begin{frame}{Objective 1: Justification for a Single Primary Mode}
    \textbf{Evidence for a Single High-Frequency Instability}
    \begin{itemize}
        \item \textbf{Unified poloidal structure:} the \SI{80.0}{\kilo\hertz}--\SI{120.0}{\kilo\hertz} band contains only one dominant poloidal mode ($m = +3$ at \SI{89.84}{\kilo\hertz}).
        \item \textbf{Frequency sweeping (chirping):} the analysis flags a frequency-sweep interval and isolates a flat-frequency sub-window for further processing. The apparent \SI{95.5}{\kilo\hertz} detection is most likely the spectral footprint of the \SI{89.0}{\kilo\hertz} mode drifting in frequency during the discharge, rather than a separate instability.
        \item \textbf{Sensitivity to the search boundary:} closely spaced spectral peaks can appear or disappear depending on how the search window is defined, so such detections are not necessarily distinct physical modes.
        \item On this basis, the \SI{89.0}{\kilo\hertz} signal is taken as the primary mode for the $n = -5$ toroidal fit and radial localization that follow.
    \end{itemize}
\end{frame}

% Slide: Obj 1 Spatial & Radial Localization
\begin{frame}{Objective 1: Spatial and Radial Localization}
    \textbf{Structural Profile of the \SI{89.0}{\kilo\hertz} Target Mode}
    \begin{itemize}
        \item The best-fit toroidal mode number was identified as $n = -5$ (aliased as $n = -1$) with a minimum Root Mean Square Error (RMSE) of \SI{6.6}{\degree}.
        \item Radial fluctuation amplitudes peaked near the plasma edge/core boundary on the ECE13FAST channel (\SI{66.5}{\giga\hertz}) reaching a Root Mean Square (RMS) of \SI{0.478}{\volt}.
        \item Maximum radial coherence was observed deeper within the plasma on the ECE03FAST channel (\SI{61.5}{\giga\hertz}) with $\gamma^2 = 0.187$.
        \item Edge fluctuation coupling was confirmed dominantly at HAFAST7.5 ($\gamma^2 = 0.2335$, RMS = \SI{1.25e-4}{\volt}), demonstrating core-edge mode linkage.
    \end{itemize}
\end{frame}

\begin{frame}{Objective 1: Comparative Analysis --- ECH+NBI vs. NBI-only}
    \textbf{Extending Mode Identification Across Heating Regimes (Shot \#88653)}
    \begin{itemize}
        \item ECH is disconnected at $t = \SI{291.300}{\milli\second}$. Two windows are compared: \textbf{ECH+NBI} (\SIrange{259.1}{291.3}{\milli\second}, \SI{32.2}{\milli\second} duration) and \textbf{NBI-only} (\SIrange{291.3}{300.0}{\milli\second}, \SI{8.7}{\milli\second} duration).
        \item \textit{Caveat 1:} The NBI beams turn off at $\sim\SI{300.5}{\milli\second}$. Any dynamic features analyzed past \SI{300.0}{\milli\second} (e.g., sliding-window traces) straddle this turn-off and extend into the fast-ion slowing-down phase, and are thus not strictly "NBI-only."
        \item \textit{Caveat 2:} The \SIrange{291.3}{300.0}{\milli\second} window is roughly $3.7\times$ shorter than the ECH phase, so it yields fewer independent segments for statistical estimators — absolute magnitudes should be compared with this in mind.
        \item The same dual-criterion ($3\sigma$ amplitude + $\gamma^2 > 0.7$) pipeline is re-applied independently to each window.
    \end{itemize}
\end{frame}

\begin{frame}{Objective 2: Dual-Band Dynamic Structure}
    \textbf{Separation of MHD Phenomena (Shot \#88653)}

        \item \textbf{Low-Frequency Band (\SI{40.0}{\kilo\hertz} -- \SI{80.0}{\kilo\hertz}):}
        \begin{itemize}
            \item Dominant poloidal mode identified as $m = +2$ at \SI{42.97}{\kilo\hertz}.
            \item The optimal physical model fit is the "Current-induced ($I_p$)" model, yielding a strong correlation of $|r| = 0.8465$.
            \item Theoretical Alfvén scaling validation returned a negative correlation ($r = -0.439$).
        \end{itemize}
        \item \textbf{High-Frequency Band (\SI{80.0}{\kilo\hertz} -- \SI{120.0}{\kilo\hertz}):}
        \begin{itemize}
            \item Dominant poloidal mode identified as $m = +3$ at \SI{89.84}{\kilo\hertz}.
            \item Best-fitting energetic particle/MHD behavior aligns with the "BAE Acoustic / GAM ($\sqrt{T_e}$)" model ($|r| = 0.4789$).
            \item Alfvén scaling validation remains weak ($r = 0.161$).
        \end{itemize}
    \end{itemize}

% Slide: Obj 3 Bicoherence interpretation, part 1 -- derivation (per bispectral_analysis_conclusion.md)
\begin{frame}{Objective 3: Origin of the $-0.5$ Bicoherence Slope}
    \textbf{From the Resonant Triad Condition to the Observed Locus}
    \begin{itemize}
        \item Three-wave coupling requires the resonant matching condition $f_3 = f_1 + f_2$.
        \item The leading triad clusters along $f_2 \approx -0.5\,f_1$, peaking at $f_1 = \SI{98.6}{\kilo\hertz}$, $f_2 = \SI{-48.8}{\kilo\hertz}$. Substituting into the matching condition:
    \end{itemize}
    \[
        f_3 = f_1 + f_2 = f_1 - 0.5\,f_1 = 0.5\,f_1
        \quad\Longrightarrow\quad
        |f_2| = |f_3| = \tfrac{1}{2}f_1
    \]
    \begin{itemize}
        \item This is the algebraic signature of an exact $f \leftrightarrow f/2$ phase lock -- a frequency and its own half-harmonic.
        \item The geometry alone does not say \emph{why} $f_1$ and $f_1/2$ are locked; that requires a physical mechanism.
    \end{itemize}
\end{frame}

% Slide: Obj 3 Bicoherence interpretation, part 2 -- four candidate mechanisms (table)
\begin{frame}{Objective 3: Four Candidate Mechanisms for the $f \leftrightarrow f/2$ Lock}
    \tiny
    \renewcommand{\arraystretch}{1.1}
    \setlength{\tabcolsep}{4pt}
    \begin{tabularx}{\textwidth}{@{}p{1.9cm}XXX@{}}
        \toprule
        \textbf{Mechanism} & \textbf{Core idea} & \textbf{Support / precedent} & \textbf{Fit to this locus} \\
        \midrule
        Harmonic self-coupling &
        Chirping mode at $f_0$ produces $2f_0$; read as pump ($f_1$), $f_0$ as $f_2=-0.5f_1$. &
        Arises \emph{by construction} from one oscillator -- no second wave needed. &
        Consistent with \SI{89.0}{\kilo\hertz} mode's observed chirping; simplest option. \\
        \addlinespace
        Degenerate PDI &
        Pump at $f_1$ decays into two \emph{identical} daughters, $f_1 \rightarrow f_1/2+f_1/2$. &
        Non-degenerate AE$\to$GAM/kTAE decay has theory support (Chen \& Zonca, 2016); non-degenerate Alfv\'enic/EP decay bicoherence-confirmed on HL-2A. &
        \emph{Degenerate} case established mainly in microwave/upper-hybrid plasmas -- different regime; treated as analogy, not direct precedent. \\
        \addlinespace
        Period-doubling bifurcation &
        Limit cycle bifurcates past an amplitude threshold, spontaneously generating $f_1/2$ -- first Feigenbaum step to chaos. &
        Seen in MHD dynamo sims (Taylor--Green forcing) and double-layer dynamics in lab plasmas. &
        Phase-locked to $f_1$ by construction, same locus; not yet reported in a fusion-MHD bicoherence context. \\
        \addlinespace
        EPM/BAE sidebands (Berk--Breizman) &
        Explosive saturation of an EP-driven mode splits energy into spectral components as holes/clumps form. &

    \end{tabularx}

\section{Comparative Conclusions}

\begin{frame}{Comparative Conclusions: Frequency Downshift at ECH Shut-off}
    \begin{itemize}
        \item \textbf{Primary mode:} drops from $\sim$89.0--93.7 kHz (ECH+NBI) to $\sim$84.5--86.4 kHz (NBI-only), $\Delta f \sim -4.5$ to $-7.3$ kHz.
        \item \textbf{Secondary mode:} drops from $\sim$47.4--50.8 kHz (ECH+NBI) to $\sim$35.2--40.5 kHz (NBI-only), $\Delta f \sim -10.0$ to $-12.2$ kHz.
        \item \textbf{Candidate cause 1 -- Thermal electron cooling:} BAE/EPM eigenfrequencies scale with sound speed, $\omega_{\text{BAE}} \sim (c_s/R)\sqrt{7/4 + T_i/T_e}$. Loss of \SI{53.2}{\giga\hertz} ECH power at $t=\SI{291.3}{\milli\second}$ cools $T_e$, lowering $c_s$ and shifting the geodesic acoustic continuum gap downward.
        \item \textbf{Candidate cause 2 -- Rotation/Doppler shift:} loss of ECH alters the radial electric field $E_r$ and bulk rotation $v_\phi$, changing the Doppler contribution $\omega_{\text{lab}} = \omega_{\text{plasma}} + n\,\omega_{\text{rot}}$.
    \end{itemize}
\end{frame}

\begin{frame}{Comparative Conclusions: Persistence of Structure \& Dynamics}
    \begin{itemize}
        \item \textbf{Continuity vs.\ bursty dynamics:} $3\sigma$ dominance occupancy is 100\% in both windows -- the mode is continuously sustained by the NBI fast-ion drive, not decaying. Both windows instead show quasi-discrete frequency hopping among 2--3 branches (jump rates $\sim$0.4--1.3/ms), a hallmark of non-linear energetic-particle-mode (EPM)/BAE avalanche and hole-clump behavior in fast-ion phase space.
        \item \textbf{Spatial structure:} the confirmed high-frequency mode retains poloidal number $m = +3$ and toroidal number $n = -5$ (aliased $n=-1$) across both windows -- same helicity before and after ECH shut-off.
    \end{itemize}
\end{frame}

\begin{frame}{Comparative Conclusions: Non-linear Coupling \& Self-Modulation}
    \begin{itemize}
        \item Strong bicoherence ($b^2 > 0.60$) connects the secondary mode ($f_2 \sim 40$--50 kHz) and primary mode ($f_1 \sim 85$--90 kHz) via the sum triad $f_3 \sim 125$--140 kHz.
        \item Self-coupling coherence shows clear low-frequency modulation peaks at $f_{\text{mod}} \sim 1.2$--$2.2$ kHz in both regimes, supporting a real non-linear wave-envelope coupling.
        \item Taken together with Objective 3's double-validation result, these non-linear signatures are statistically present but \textbf{not yet independently confirmed} as a specific physical mechanism (e.g., PDI) -- Harmonic Self-Coupling remains the most defensible interpretation pending cross-diagnostic corroboration.
    \end{itemize}
\end{frame}
