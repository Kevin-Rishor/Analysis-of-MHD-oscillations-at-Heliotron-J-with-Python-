\section{Motivation \& Objectives}
% ============================================================

% SLIDE 2 -- Motivation + all three objectives, one line each (~60s)
\begin{frame}{Motivation \& Objectives}
    \textbf{Why this matters:} MHD instabilities driven by energetic particles and
    low-frequency H-mode fluctuations degrade confinement in helical devices like
    Heliotron J, causing fast-ion loss and enhanced edge heat transport.

    \vspace{0.3cm}
    \textbf{Three specific objectives, one shot (\#88653):}
    \begin{enumerate}
        \item \textbf{Identify \& validate} the dominant coherent MHD mode(s)
        (dual criterion: $3\sigma$ amplitude \emph{and} $\gamma^2>0.7$ spatial coherence),
        determine its toroidal number $n$, and localize it radially.
        \item \textbf{Separate} low- and high-frequency bands, test them against
        physical scaling models, and quantify non-linear self-coupling.
        \item \textbf{Detect and rigorously validate} non-linear three-wave
        (bicoherence) coupling tied to H-mode transport.
    \end{enumerate}

    \vspace{0.2cm}
    \textbf{Running thread:} every objective is repeated for the
    \textbf{ECH+NBI} and \textbf{NBI-only} phases of the same discharge, to test
    whether what we find survives the loss of ECH heating.
\end{frame}

% ============================================================
\section{Methodology}
% ============================================================

% SLIDE 3 -- Method in one slide (~45s)
\begin{frame}{Method: Dual-Criterion Validation \& Comparative Windows}
    \begin{itemize}
        \item \textbf{Diagnostics:} toroidal Mirnov array (MP1/MP3/MP4), poloidal
        array (PMP1--PMP14), ECE electron-temperature channels, diamagnetic loop
        ($W_p$), fast H$\alpha$.
        \item \textbf{Time-frequency:} STFT spectrogram + Morlet continuous
        wavelet transform.
        \item \textbf{Dual-criterion gate:} a peak is only accepted as a
        \emph{physical} mode if it clears a $3\sigma$ noise threshold \emph{and}
        shows spatial coherence $\gamma^2>0.7$ on at least one coil pair --
        amplitude alone is never enough.
        \item \textbf{Comparative design:} ECH switches off at $t=\SI{291.3}{ms}$
        in shot \#88653. We compare \textbf{ECH+NBI} (\SIrange{259.1}{291.3}{ms})
        vs.\ \textbf{NBI-only} (\SIrange{291.3}{300.0}{ms}; NBI beams turn off
        at ${\sim}\SI{300.6}{ms}$) -- same shot, same pipeline, run twice,
        isolating the effect of ECH loss.
    \end{itemize}
\end{frame}

\section{Objective 3: Non-linear Coupling (Bicoherence)}
% ============================================================

% SLIDE 10 -- Obj 3 discovery (~90s)
\begin{frame}{Objective 3: A Striking $f \leftrightarrow f/2$ Locking Signature}
    \begin{figure}[h!]
        \centering
        \includegraphics[height=0.5\textheight]{Images/mhd_bicoherence_objective3_88653.png}
    \end{figure}
    \footnotesize
    Auto-bicoherence of MP1 (mode-active window, \SIrange{259.1}{275.0}{ms})
    peaks at $b^2=0.515$ for the triad $f_1=\SI{98.6}{kHz}$,
    $f_2=\SI{-48.8}{kHz}$, with the leading triads clustering along
    $f_2\approx-0.5f_1$. Substituting into the resonance condition
    $f_3=f_1+f_2$:
    \[
        f_3 = f_1 - 0.5f_1 = 0.5f_1 \quad\Longrightarrow\quad |f_2|=|f_3|=\tfrac12 f_1
    \]
    -- the algebraic signature of an \textbf{exact $f \leftrightarrow f/2$ phase lock}.
\end{frame}

% SLIDE 11 -- Obj 3 energy-transfer direction (NEW) (~80s)
\begin{frame}{Objective 3: Energy Flows One Way -- Coalescence, Not Decay}
    \begin{figure}[h!]
        \centering
        % IMAGE TO ADD: time-trace of the corrected biphase / Im[B_sum] sign
        % along the -0.5 subharmonic ridge for the valid high-SNR windows.
        % See chat message for exact plotting spec and suggested filename.
        \includegraphics[height=0.32\textheight]{Images/mhd_biphase_direction_shot_88653.png}
    \end{figure}
    \footnotesize
    \begin{itemize}
        \item The \textbf{sign} of $\mathrm{Im}[B_{\text{sum}}]$ sets the
        energy-transfer direction: $>0$ = \textbf{coalescence} (flows up);
        $<0$ = decay instability (flows down).
        \item Correcting for window overlap lowers the effective ensemble
        count to $N_{\text{eff}}=15$, raising the proper 95\% threshold to
        $b^2_{95}\approx0.20$ -- double the uncorrected value used elsewhere.
        \item The $-0.5$ subharmonic ridge (strongest in the NBI-only phase,
        $b^2=0.88$) clears this stricter threshold by $>4\times$.
        \item After excluding low-SNR samples, \textbf{all 16} valid high-SNR
        measurements (two episodes: 6 windows at 295.3--297.1\,ms during NBI,
        10 windows at 301.8--304.6\,ms during post-NBI slowing-down) show
        $\mathrm{Im}[B_{\text{sum}}]>0$ -- the direction never reverses.
    \end{itemize}
\end{frame}

        \item Amplitude scaling: OLS slope ${\approx}\,2.1$ (band-integrated)

        real, unidirectional, avalanche-like character is the well-supported
        result.
    \end{itemize}
\end{frame}

% SLIDE 13 -- Obj 3 validation status (~80s)
\begin{frame}{Objective 3: Real Coupling, Not Yet a Confirmed Mechanism}
    \footnotesize
    \begin{itemize}
        \item \textbf{What is confirmed:} asymptotic convergence ($1/N\to0$)
        gives a non-zero intercept ($b^2_\infty\approx0.28$--$0.33$ for the
        leading triads vs.\ $\approx0$ for the noise baseline) -- a
        \textbf{real} quadratic phase coupling, not a finite-sample artifact.
        \item \textbf{What fails:} strict double-validation --
        shot-to-shot reproducibility could not be tested (validation shot
        \#88654 data unavailable), and a scan of 36 Langmuir probe tips found
        \textbf{zero} matches above $b^2>0.5$.
        \item \textbf{Self-critical check:} the burst window is \emph{not}
        statistically superior to quiescent (max $b^2=0.37$) or control
        (max $b^2=0.46$) periods -- no rigorous basis to call this an
        exclusive H-mode signature.
        \item \textbf{Comparative:} NBI-only triads reach up to $\sim$3$\times$
        higher $b^2$ than ECH+NBI -- coupling appears to strengthen once ECH
        switches off, but the same validation caveat applies.
        \item \textbf{Conclusion:} statistically real and unidirectional, but
        \emph{not yet independently confirmed} as a specific mechanism --
        reported honestly as an open finding.
    \end{itemize}
\end{frame}

\section{Conclusions}
% ============================================================

% SLIDE 14 -- Conclusions (~85s)
\begin{frame}{Conclusions}
    \footnotesize
    \begin{itemize}
        \item \textbf{Objective 1:} A single $m=+3$, $n=-5$ energetic-particle
        mode at $\sim$\SI{89}{kHz} is robustly identified, localized at the
        edge/core boundary, and \textbf{keeps its identity} across ECH
        shut-off -- only its frequency shifts.
        \item \textbf{Objective 2:} The LF and HF bands follow
        \textbf{different physics} (current-driven vs.\ BAE/GAM-like), are
        dominated by the \textbf{NBI drive} rather than ECH, and are
        non-linearly coupled both to each other and to their own envelopes.
        \item \textbf{Objective 3:} The exact $f\leftrightarrow f/2$
        bicoherence locking is \textbf{statistically real and
        unidirectional} (energy consistently flows upward, never downward)
        -- but \emph{which} mode is the true instability, and the underlying
        mechanism, remain \textbf{open questions} pending stronger
        cross-diagnostic evidence.
        \item \textbf{Confinement relevance:} the primary mode's edge/core
        localization and confirmed edge-H$\alpha$ coupling position it as a
        plausible fast-ion/heat-loss channel, though a direct transport
        coefficient has not yet been quantified -- a natural next step.
    \end{itemize}
    \vspace{0.2cm}
    \textbf{Next steps:} re-analysis with a working validation shot, a
    refined Langmuir cross-check, phase-locking tests to resolve mode
    identity, and extension of this pipeline to a broader shot database.
\end{frame}
